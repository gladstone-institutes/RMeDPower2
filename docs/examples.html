Real-World Examples

Biomedical research applications of RMeDPower2

1. Cell Assays - ALS iPSC Study

Analysis of iPSC-derived motor neurons from ALS patients and controls. This example demonstrates hierarchical experimental design with nested factors.

Study Design

Hierarchy: Experiment → Cell Line → Individual Cells
Response: Soma size of motor neurons
Treatment: Control vs. sALS patient lines

library(RMeDPower2)

# Load cell assay data
data <- read.csv("cell_size_data.csv", header = TRUE)

# Define experimental design
design <- new("RMeDesign",
    response_column = "cell_size2",
    condition_column = "classification",  # Control vs sALS
    experimental_columns = c("experiment", "line"),
    condition_is_categorical = TRUE,
    crossed_columns = "line"  # Cell lines appear across experiments
)

# Normal distribution assumption
model <- new("ProbabilityModel",
    error_is_non_normal = FALSE
)

# Power parameters for experiment scaling
power_param <- new("PowerParams",
    target_columns = c("experiment"),
    power_curve = 1,
    nsimn = 100,
    levels = c(1)
)

# Step 1: Diagnose data
diagnose_res <- diagnoseDataModel(data, design, model)

# Check for outlier experiments
print(diagnose_res$cooks_distance_experimental_column1)

# Check for outlier cell lines  
print(diagnose_res$cooks_distance_experimental_column2)

# Step 2: Use log-transformed data without outliers
data_updated <- diagnose_res$Data_updated
design_updated <- design
design_updated@response_column <- "cell_size2_logTransformed_noOutlier"

# Step 3: Estimate parameters
estimate_res <- getEstimatesOfInterest(data_updated, design_updated, model)
print(estimate_res[[1]])

# Step 4: Calculate power for different numbers of experiments
power_res <- calculatePower(data_updated, design_updated, model, power_param)

2. Single-Cell RNA-seq - APOE Study

Analysis of single-nuclei RNA-seq data examining APOE4 effects in Alzheimer's disease mouse models. Demonstrates both categorical and continuous predictors with count data.

Cell Type Proportions Analysis

Design: Mouse → Cell counts per cluster
Distribution: Binomial (proportions)
Comparison: APOE4-KI vs APOE4-KI-Syn1-Cre

# Cell type proportion analysis
data <- read.csv("apoe_associated_cell_proportions.csv", header = TRUE)

# Design for binomial model (proportions)
design <- new("RMeDesign",
    response_column = "cluster7_count",
    condition_column = "genotype",
    experimental_columns = c("animal_model", "sample_id"),
    total_column = "total_nuclei",  # For binomial offset
    condition_is_categorical = TRUE
)

# Binomial distribution for proportions
model <- new("ProbabilityModel",
    error_is_non_normal = TRUE,
    family_p = "binomial"
)

# Power analysis for number of mice
power_param <- new("PowerParams",
    target_columns = c("sample_id"),
    power_curve = 1,
    nsimn = 100,
    levels = c(1)
)

# Workflow
diagnose_res <- diagnoseDataModel(data, design, model)
estimate_res <- getEstimatesOfInterest(data, design, model)
power_res <- calculatePower(data, design, model, power_param)

Continuous Predictor: Hippocampus Volume

Testing association between cell proportions and brain volume (continuous predictor)

# Modify design for continuous predictor
design@condition_column <- "Hippocampus_Vol_mm"
design@condition_is_categorical <- FALSE

# Same analysis workflow
diagnose_res <- diagnoseDataModel(data, design, model)
estimate_res <- getEstimatesOfInterest(data, design, model)
power_res <- calculatePower(data, design, model, power_param)

Gene Expression Analysis

Count-based analysis of individual gene expression

# Gene expression counts
data <- read.csv("raw_counts_Snhg11_gene_cluster_7.csv", header = TRUE)

design <- new("RMeDesign",
    response_column = "raw_counts",
    condition_column = "genotype", 
    experimental_columns = c("animal_model", "sample_id", "cell_id"),
    total_column = "library_size",  # Sequencing depth offset
    condition_is_categorical = TRUE
)

# Negative binomial for count data
model <- new("ProbabilityModel",
    error_is_non_normal = TRUE,
    family_p = "negative_binomial"
)

# Power analysis with known effect size
power_param@effect_size <- log(2)  # 2-fold change
power_res <- calculatePower(data, design, model, power_param)

3. Behavioral Assays - Morris Water Maze

Spatial learning assessment in mouse models. Demonstrates repeated measures with potential log transformation for skewed response times.

Study Design

Hierarchy: Mouse → Trial
Response: Latency to reach platform (seconds)
Treatment: Disease model vs. Wild-type

# Morris Water Maze data
data <- read.csv("MWM_data.csv", header = TRUE)

design <- new("RMeDesign",
    response_column = "latency",
    condition_column = "genotype",
    experimental_columns = c("MouseID", "trial"),
    condition_is_categorical = TRUE
)

model <- new("ProbabilityModel",
    error_is_non_normal = FALSE
)

# Power analysis for number of mice
power_param <- new("PowerParams",
    target_columns = c("MouseID"),
    power_curve = 1,
    nsimn = 100,
    levels = c(1)
)

# Diagnose - likely need log transformation
diagnose_res <- diagnoseDataModel(data, design, model)

# Use log-transformed data
data_updated <- diagnose_res$Data_updated
design_updated <- design
design_updated@response_column <- "latency_logTransformed"

# Final analysis
estimate_res <- getEstimatesOfInterest(data_updated, design_updated, model)
power_res <- calculatePower(data_updated, design_updated, model, power_param)

4. Electrophysiology - Patch Clamp

Membrane potential recordings from brain slices. Shows deeply nested experimental design with multiple levels of biological replication.

Study Design

Hierarchy: Mouse → Brain Slice → Recorded Cell
Response: Membrane action potential (mV)
Treatment: hAPP disease model vs. Wild-type

# Patch clamp data
data <- read.csv("patch_clamp_data.csv", header = TRUE)

design <- new("RMeDesign",
    response_column = "membrane_potential",
    condition_column = "genotype",
    experimental_columns = c("Mice", "slice_number", "cell_id"),
    condition_is_categorical = TRUE
)

model <- new("ProbabilityModel",
    error_is_non_normal = FALSE
)

# Test power for different experimental scales
# Option 1: More mice
power_param_mice <- new("PowerParams",
    target_columns = c("Mice"),
    power_curve = 1,
    effect_size = 7,  # Expected difference in mV
    nsimn = 100
)

# Option 2: More slices per mouse
power_param_slices <- new("PowerParams",
    target_columns = c("slice_number"), 
    power_curve = 1,
    effect_size = 7,
    nsimn = 100
)

# Workflow
diagnose_res <- diagnoseDataModel(data, design, model)
estimate_res <- getEstimatesOfInterest(data, design, model)

# Compare power strategies
power_mice <- calculatePower(data, design, model, power_param_mice)
power_slices <- calculatePower(data, design, model, power_param_slices)

5. Advanced Simulation Scenarios

Multi-level Power Analysis

Testing power across multiple experimental variables simultaneously

# Test both experiments and cell lines
power_param <- new("PowerParams",
    target_columns = c("experiment", "line"),
    power_curve = 1,
    levels = c(1, 1),  # Increase both levels and samples
    nsimn = 100
)

power_res <- calculatePower(data, design, model, power_param)

Sample Size vs. Replication Trade-offs

Compare increasing experimental units vs. samples per unit

# Strategy 1: More experimental units
power_param1 <- power_param
power_param1@levels <- c(1)  # level = 1

# Strategy 2: More samples per unit  
power_param2 <- power_param
power_param2@levels <- c(0)  # level = 0

# Compare strategies
power_res1 <- calculatePower(data, design, model, power_param1)
power_res2 <- calculatePower(data, design, model, power_param2)

Using ICC When Variance Components Cannot Be Estimated

When pilot data has limited experimental structure

# Limited pilot data (single experiment, single cell line)
data_limited <- RMeDPower_data2

# Provide ICC values manually
power_param@icc <- c(0.8, 0.05, 0.05)  # High, medium, low ICC

power_res <- calculatePower(data_limited, design, model, power_param)

Tips for Success

🔍 Always Diagnose First

Model assumptions matter! Use diagnoseDataModel() before power analysis

📊 Check Residual Plots

QQ plots and residual plots reveal if transformations are needed

🎯 Remove Outliers Carefully

Consider both observation-level and group-level outliers

⚖️ Balance Power Strategies

Compare increasing units vs. samples per unit for cost-effectiveness
